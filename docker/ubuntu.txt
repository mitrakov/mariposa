# Cluster Setup

# on Mac:
ssh-keygen -t rsa -C "mariposa" -f ~/.ssh/mariposa/id_rsa
ssh-copy-id -i ~/.ssh/mariposa/id_rsa.pub hadoop@192.168.1.2
scp    -i ~/.ssh/mariposa/id_rsa ~/.ssh/mariposa/id_rsa* hadoop@192.168.1.2:~/.ssh/
scp -r -i ~/.ssh/mariposa/id_rsa docker2bash.sh hadoop.dockerfile mariposa-hbase-patch-2.5.14.jar krb/dockerfile krb/autorun/ hadoop@192.168.1.2:~



# on bastion:
for i in 02 03 04 05 06 07 08 09 10 11; do scp ~/.ssh/id_rsa* hadoop@n$i:~/.ssh/; done
for i in 02 03 04 05 06 07 08 09 10 11; do scp docker2bash.sh hadoop.dockerfile mariposa-hbase-patch-2.5.14.jar krb/dockerfile krb/autorun/ hadoop@n$i:~; done



# Every node setup
hostnamectl set-hostname nXX


cat << TOM | sudo tee -a /etc/hosts
10.1.1.1    n01
10.1.1.2    n02
10.1.1.3    n03
10.1.1.4    n04
10.1.1.5    n05
10.1.1.6    n06
10.1.1.7    n07
10.1.1.8    n08
10.1.1.9    n09
10.1.1.10   n10
10.1.1.11   n11
TOM


sudo ./docker2bash.sh hadoop.dockerfile
sudo ./docker2bash.sh dockerfile


# replace ZK_ID=2,3,4...
cat << EOF | sudo tee --append /etc/profile.d/mariposa.sh
export MASTER_HOST=n01
export WORKER_HOSTS=n02,n03,n04,n05,n06,n07,n08,n09,n10,n11
export ZK_ID=N
EOF


# autostart
crontab -e
@reboot java -jar /home/hadoop/autorun/ssu.jar /home/hadoop/autorun/ssu.json > /home/hadoop/ssu.log 2>&1

reboot
